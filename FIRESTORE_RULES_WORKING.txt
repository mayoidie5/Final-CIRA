rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // Helper function to check if a user is an admin
    // This function safely checks if a user document exists and has role == 'admin'
    function isAdmin(userId) {
      return get(/databases/$(database)/documents/users/$(userId)).data.role == 'admin';
    }

    // Helper function to check if a user is a class representative
    function isClassRep(userId) {
      let userRole = get(/databases/$(database)/documents/users/$(userId)).data.role;
      return userRole == 'class_rep' || userRole == 'class-representative';
    }

    // Rules for the 'users' collection
    match /users/{userId} {
      allow create: if request.auth != null && request.auth.uid == userId;
      allow read, update: if request.auth != null && request.auth.uid == userId;
      allow read, write: if request.auth != null && isAdmin(request.auth.uid);
    }

    // Rules for the 'tickets' collection
    match /tickets/{ticketId} {
      // All authenticated users can read all tickets
      allow read: if request.auth != null;
      
      // Any authenticated user can create a ticket
      allow create: if request.auth != null;
      
      // Ticket creator can update their own tickets
      allow update: if request.auth != null && request.auth.uid == resource.data.userId;
      
      // Ticket creator can delete their own tickets
      allow delete: if request.auth != null && request.auth.uid == resource.data.userId;
      
      // Admins and class reps can update/delete any ticket
      allow update, delete: if request.auth != null && (isAdmin(request.auth.uid) || isClassRep(request.auth.uid));
    }

    // Rules for the 'form-structure' collection
    match /form-structure/{fieldId} {
      allow read: if request.auth != null;
      allow create, update, delete: if request.auth != null && isAdmin(request.auth.uid);
    }

    // Rules for the 'notifications' collection
    match /notifications/{notificationId} {
      // Users can read their own notifications
      allow read: if request.auth != null && (request.auth.uid == resource.data.userId || isAdmin(request.auth.uid));
      
      // Users can update their own notifications (e.g., mark as read)
      allow update: if request.auth != null && (request.auth.uid == resource.data.userId || isAdmin(request.auth.uid));
      
      // Any authenticated user can create notifications (for system notifications)
      allow create: if request.auth != null;
      
      // Users can delete their own notifications
      allow delete: if request.auth != null && (request.auth.uid == resource.data.userId || isAdmin(request.auth.uid));
    }

    // Rules for the 'formConfigs' collection (for form editor)
    // Collection name is 'formConfigs' with 's' at the end
    match /formConfigs/{configId} {
      // All authenticated users can read form configs
      allow read: if request.auth != null;
      
      // Only admins can create, update, delete form configs
      allow create, update, delete: if request.auth != null && isAdmin(request.auth.uid);
    }
    
    // Catch-all deny rule for any other paths
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
