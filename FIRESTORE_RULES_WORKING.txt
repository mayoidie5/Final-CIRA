rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // Rules for the 'users' collection
    match /users/{userId} {
      allow create: if request.auth != null && request.auth.uid == userId;
      // Allow reading own user or any authenticated user
      allow read: if request.auth != null;
      // Allow any authenticated user to update users (for email verification from anonymous auth)
      allow update: if request.auth != null;
      // Allow deletion only by own user
      allow delete: if request.auth != null && request.auth.uid == userId;
    }

    // Rules for the 'tickets' collection - PERMISSIVE FOR TESTING
    match /tickets/{ticketId} {
      // All authenticated users can read all tickets
      allow read: if request.auth != null;
      
      // Any authenticated user can create a ticket
      allow create: if request.auth != null;
      
      // Any authenticated user can update any ticket
      allow update: if request.auth != null;
      
      // Any authenticated user can delete any ticket
      allow delete: if request.auth != null;
      
      // Sub-collection for comments
      match /comments/{commentId} {
        allow read: if request.auth != null;
        allow create, update, delete: if request.auth != null;
      }
    }

    // Rules for the 'form-structure' collection
    match /form-structure/{fieldId} {
      allow read, write: if request.auth != null;
    }

    // Rules for the 'notifications' collection
    match /notifications/{notificationId} {
      allow read, write, create, update, delete: if request.auth != null;
    }

    // Rules for the 'formConfigs' collection
    match /formConfigs/{configId} {
      allow read, write, create, update, delete: if request.auth != null;
    }

    // Rules for the 'verificationTokens' collection - ALLOW PUBLIC ACCESS FOR EMAIL VERIFICATION
    match /verificationTokens/{document=**} {
      allow read, write, create, update, delete: if true;
    }
    
    // Catch-all deny rule for any other paths
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
